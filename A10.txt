Step 1: Install Prerequisites
sudo apt update
sudo apt install curl git docker.io docker-compose -y

Check Docker:

docker --version
docker compose version
✅ Step 2: Download Hyperledger Fabric Samples
curl -sSL https://bit.ly/2ysbOFE | bash -s

This downloads:

fabric-samples
binaries
docker images
✅ Step 3: Go to Test Network Folder
cd fabric-samples/test-network
✅ Step 4: Start Blockchain Network
./network.sh up

This creates:

2 Organizations
Peer nodes
Ordering service
CA
✅ Step 5: Create a Channel
./network.sh createChannel

Channel name usually:

mychannel
✅ Step 6: Deploy Smart Contract

Use basic asset-transfer chaincode.

./network.sh deployCC -ccn basic -ccp ../asset-transfer-basic/chaincode-go -ccl go

If using JavaScript:

./network.sh deployCC -ccn basic -ccp ../asset-transfer-basic/chaincode-javascript -ccl javascript
✅ Step 7: Set Environment Variables
export PATH=${PWD}/../bin:$PATH
export FABRIC_CFG_PATH=$PWD/../config/
export CORE_PEER_TLS_ENABLED=true
export CORE_PEER_LOCALMSPID="Org1MSP"
export CORE_PEER_TLS_ROOTCERT_FILE=${PWD}/organizations/peerOrganizations/org1.example.com/peers/peer0.org1.example.com/tls/ca.crt
export CORE_PEER_MSPCONFIGPATH=${PWD}/organizations/peerOrganizations/org1.example.com/users/Admin@org1.example.com/msp
export CORE_PEER_ADDRESS=localhost:7051
✅ Step 8: Initialize Ledger
peer chaincode invoke -o localhost:7050 \
--ordererTLSHostnameOverride orderer.example.com \
--tls \
--cafile "${PWD}/organizations/ordererOrganizations/example.com/orderers/orderer.example.com/msp/tlscacerts/tlsca.example.com-cert.pem" \
-C mychannel -n basic \
-c '{"function":"InitLedger","Args":[]}'
✅ Step 9: Query Blockchain Data
peer chaincode query -C mychannel -n basic -c '{"Args":["GetAllAssets"]}'
✅ Step 10: Add New Asset
peer chaincode invoke -o localhost:7050 \
--ordererTLSHostnameOverride orderer.example.com \
--tls \
--cafile "${PWD}/organizations/ordererOrganizations/example.com/orderers/orderer.example.com/msp/tlscacerts/tlsca.example.com-cert.pem" \
-C mychannel -n basic \
-c '{"function":"CreateAsset","Args":["asset9","red","10","Tushar","500"]}'
✅ Step 11: Query Again
peer chaincode query -C mychannel -n basic -c '{"Args":["ReadAsset","asset9"]}'
✅ Step 12: Stop Network
./network.sh down
✅ Output for Practical Viva

This creates a permissioned blockchain because:

Only authorized members join network
Identities managed using MSP
Transactions validated by peers
Immutable ledger maintained
