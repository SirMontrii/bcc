OUTPUT: nano sign.cpp , g++ sign.cpp -o sign -lssl -lcrypto , ./sign 
Code:
#include <iostream>
#include <string>
#include <iomanip>
#include <sstream>
#include <openssl/sha.h>

using namespace std;

// SHA256 hash function
string sha256(const string &input) {
    unsigned char hash[SHA256_DIGEST_LENGTH];

    SHA256((unsigned char*)input.c_str(), input.length(), hash);

    stringstream ss;
    for (int i = 0; i < SHA256_DIGEST_LENGTH; i++) {
        ss << hex << setw(2) << setfill('0') << (int)hash[i];
    }

    return ss.str();
}

int main() {
    string document;

    cout << "Enter document/message: ";
    getline(cin, document);

    cout << "\nOriginal Document: " << document << endl;

    // Signing
    string signature = sha256(document);

    cout << "Digital Signature: " << signature << endl;

    // Verification
    string receivedDocument;

    cout << "\nEnter received document for verification: ";
    getline(cin, receivedDocument);

    string verifyHash = sha256(receivedDocument);

    if (verifyHash == signature) {
        cout << "Verification Successful: Signature is VALID." << endl;
    } else {
        cout << "Verification Failed: Signature is INVALID." << endl;
    }

    return 0;
}
